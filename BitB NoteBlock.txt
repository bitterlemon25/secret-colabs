def try_download(url, out):
    subprocess.run(['wget', '-q', '-L', url, '-O', out])
    result = subprocess.run(['file', out], capture_output=True, text=True)
    return 'bzip2' in result.stdout

# ============ 7. Resize backend: real xrandr resize + force Firefox window to match ============
resize_server_code = '''
import subprocess, os
from http.server import BaseHTTPRequestHandler, HTTPServer
from urllib.parse import urlparse, parse_qs

ENV = {**os.environ, "DISPLAY": ":1"}

class ResizeHandler(BaseHTTPRequestHandler):
    def do_GET(self):
        parsed = urlparse(self.path)
        if parsed.path == "/resize":
            qs = parse_qs(parsed.query)
            w = qs.get("w", [None])[0]
            h = qs.get("h", [None])[0]
            if w and h:
                try:
                    w, h = int(float(w)), int(float(h))
                    subprocess.run(["xrandr", "--fb", f"{w}x{h}"], env=ENV, check=False)
                    # force firefox window to exactly cover the new screen size
                    ids = subprocess.run(
                        ["xdotool", "search", "--class", "Firefox"],
                        env=ENV, capture_output=True, text=True
                    ).stdout.split()
                    for wid in ids:
                        subprocess.run(["xdotool", "windowsize", wid, str(w), str(h)], env=ENV, check=False)
                        subprocess.run(["xdotool", "windowmove", wid, "0", "0"], env=ENV, check=False)
                    self.send_response(200)
                except Exception:
                    self.send_response(500)
            else:
                self.send_response(400)
        else:
            self.send_response(404)
        self.send_header("Access-Control-Allow-Origin", "*")
        self.end_headers()

    def log_message(self, format, *args):
        pass

HTTPServer(("127.0.0.1", 6081), ResizeHandler).serve_forever()
'''
with open('/content/resize_server.py', 'w') as f:
    f.write(resize_server_code)

def start_tunnel_and_wait(local_port, max_attempts=4, dns_wait_timeout=45):
    for attempt in range(1, max_attempts + 1):
        print(f"[tunnel:{local_port}] attempt {attempt}/{max_attempts}...")
        proc = subprocess.Popen(
            ['/content/cloudflared', 'tunnel', '--url', f'http://localhost:{local_port}'],
            stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True
        )
        url = None
        for _ in range(60):
            line = proc.stdout.readline()
            if not line:
                time.sleep(0.5)
                continue
            match = re.search(r'https://[a-zA-Z0-9\-]+\.trycloudflare\.com', line)
            if match:
                url = match.group(0)
                break
        if not url:
            proc.terminate()
            continue
        deadline = time.time() + dns_wait_timeout
        while time.time() < deadline:
            try:
                r = requests.get(url, timeout=5)
                if r.status_code < 500:
                    print(f"[tunnel:{local_port}] live: {url}")
                    return proc, url
            except requests.exceptions.RequestException:
                pass
            time.sleep(3)
        proc.terminate()
    return None, None

# Tunnel #1: resize backend (must exist before we can reference it in index.html)
resize_tunnel_proc, resize_url = start_tunnel_and_wait(6081)

<html>
<head>
<meta charset="utf-8">
<meta name="viewport" content="width=device-width, initial-scale=1, maximum-scale=1, user-scalable=no">
<title>Whatsapp</title>

  html, body {{ margin:0; padding:0; width:100%; height:100%; overflow:hidden; background:#fff; }}
  #screen {{ width:100vw; height:100dvh; overflow:hidden; }}
  canvas {{ display:block; }} /* native resolution, matches real device size — no scale, no crop */
</style>
</head>
<body>
<div id="screen"></div>
<script type="module">
  import RFB from './core/rfb.js';

  const RESIZE_BASE = "{resize_url}";
  const host = window.location.hostname;
  const port = window.location.port || (window.location.protocol === 'https:' ? 443 : 80);
  const proto = window.location.protocol === 'https:' ? 'wss' : 'ws';
  const wsUrl = proto + '://' + host + (port ? ':' + port : '') + '/websockify';

  const screen = document.getElementById('screen');
  let rfb;

  async function setServerSize() {{
    const w = Math.round(window.innerWidth);
    const h = Math.round(window.innerHeight);
    try {{
      await fetch(`${{RESIZE_BASE}}/resize?w=${{w}}&h=${{h}}`);
    }} catch (e) {{}}
  }}

  async function connect() {{
    await setServerSize();
    await new Promise(r => setTimeout(r, 900)); // let X + Firefox settle into new size
    rfb = new RFB(screen, wsUrl);
    rfb.viewOnly = false;
    rfb.scaleViewport = false;
    rfb.resizeSession = false; // we handle resizing ourselves via xrandr backend
    rfb.addEventListener('disconnect', () => setTimeout(connect, 1000));
  }}
  connect();

  let resizeTimer;
  window.addEventListener('resize', () => {{
    clearTimeout(resizeTimer);
    resizeTimer = setTimeout(async () => {{
      await setServerSize();
      if (rfb) {{ rfb.disconnect(); }} // reconnect to pick up fresh framebuffer size cleanly
    }}, 300);
  }});
</script>
</body>
</html>
"""
